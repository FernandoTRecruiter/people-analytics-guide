Turnover Rate
Benchmark: 10–15% · Fonte: SHRM
Fórmula: Turnover Rate (%) = (Saídas no período / HC Médio) × 100
HC Médio = (HC início + HC final) / 2 · Tabela: fato_Historico (Tipo_Evento = "Desligamento")
Excel — Taxa de Turnover
=CONT.SES(
    Historico[Tipo_Evento],  "Desligamento",
    Historico[Data_Evento],  ">="&DATA(ANO(HOJE()),1,1),
    Historico[Data_Evento],  "<="&HOJE()
) / ((HC_Inicio + HC_Final) / 2) * 100

Nomeie as células: B2 = HC_Inicio (headcount em 1° jan) e B3 = HC_Final (headcount atual).

DAX — Power BI
Saídas no Período =
CALCULATE(COUNTROWS(fato_Historico), fato_Historico[Tipo_Evento] = "Desligamento")

HC Médio =
DIVIDE(
    CALCULATE(COUNTROWS(dim_Colaboradores), dim_Colaboradores[Status] = "Ativo",
              DATEADD(dim_dCalendario[Date], -12, MONTH))
    + CALCULATE(COUNTROWS(dim_Colaboradores), dim_Colaboradores[Status] = "Ativo"),
    2
)

Turnover Rate % = DIVIDE([Saídas no Período], [HC Médio], 0) * 100
SQL
WITH saidas AS (
    SELECT COUNT(*) AS total_saidas
    FROM fato_Historico
    WHERE Tipo_Evento = 'Desligamento'
      AND strftime('%Y', Data_Evento) = strftime('%Y','now')
),
hc AS (SELECT COUNT(*) AS hc FROM dim_Colaboradores WHERE Status = 'Ativo')
SELECT s.total_saidas,
       ROUND(s.total_saidas * 100.0 / h.hc, 2) AS turnover_rate_pct
FROM saidas s, hc h;
Python — pandas
import pandas as pd
from datetime import date
col  = pd.read_excel("workforce.xlsx", sheet_name="Colaboradores", parse_dates=["Data_Admissao"])
hist = pd.read_excel("workforce.xlsx", sheet_name="Historico",    parse_dates=["Data_Evento"])
year = date.today().year
saidas  = hist[(hist["Tipo_Evento"]=="Desligamento") & (hist["Data_Evento"].dt.year==year)].shape[0]
hc_fim  = (col["Status"]=="Ativo").sum()
hc_ini  = col[col["Data_Admissao"].dt.year < year].shape[0]
hc_med  = (hc_ini + hc_fim) / 2
turnover = round(saidas / hc_med * 100, 2) if hc_med > 0 else 0
print(f"Turnover Rate: {turnover}%")

Benchmark: > 85% · Fonte: AIHR
Fórmula: Retention Rate (%) = 100 − Turnover Rate (%)
Ou: (Colaboradores ativos que permaneceram durante o período / HC inicial) × 100
Excel
=100 - [célula_Turnover_Rate]
DAX
Retention Rate % = 100 - [Turnover Rate %]
SQL
SELECT 100 - ROUND(
    (SELECT COUNT(*) FROM fato_Historico WHERE Tipo_Evento='Desligamento'
     AND strftime('%Y',Data_Evento)=strftime('%Y','now')) * 100.0
    / NULLIF((SELECT COUNT(*) FROM dim_Colaboradores WHERE Status='Ativo'),0)
,2) AS retention_rate_pct;
Python
retention_rate = round(100 - turnover, 2)
print(f"Retention Rate: {retention_rate}%")

eNPS (Employee Net Promoter Score)
Benchmark: 20–40 · Fonte: LinkedIn Talent Trends
Fórmula: eNPS = ((Promotores − Detratores) / Total de respondentes) × 100
Escala: Promotores = 9–10 · Neutros = 7–8 · Detratores = 0–6 · Intervalo: −100 a +100
Excel
=(CONT.SES(Avaliacoes[Tipo_Avaliacao],"Clima_Organizacional",Avaliacoes[Nota_eNPS],">="&9)
 -CONT.SES(Avaliacoes[Tipo_Avaliacao],"Clima_Organizacional",Avaliacoes[Nota_eNPS],"<="&6))
/ CONT.SE(Avaliacoes[Tipo_Avaliacao],"Clima_Organizacional") * 100
DAX — padrão correto (FILTER, não CALCULATE sobre VAR)
eNPS =
VAR base  = FILTER(fato_Avaliacoes, fato_Avaliacoes[Tipo_Avaliacao] = "Clima_Organizacional")
VAR total = COUNTROWS(base)
VAR prom  = COUNTROWS(FILTER(base, fato_Avaliacoes[Nota_eNPS] >= 9))
VAR det   = COUNTROWS(FILTER(base, fato_Avaliacoes[Nota_eNPS] <= 6))
RETURN DIVIDE(prom - det, total, 0) * 100
Atenção DAX: CALCULATE(COUNTROWS(base)) onde base é uma variável VAR não filtra corretamente. Use sempre COUNTROWS(FILTER(base, condição)).
SQL
SELECT ROUND(
    (SUM(CASE WHEN Nota_eNPS >= 9 THEN 1.0 ELSE 0 END)
   - SUM(CASE WHEN Nota_eNPS <= 6 THEN 1.0 ELSE 0 END))
    / COUNT(*) * 100, 2) AS eNPS
FROM fato_Avaliacoes WHERE Tipo_Avaliacao = 'Clima_Organizacional';
Python
aval  = pd.read_excel("workforce.xlsx", sheet_name="Avaliacoes")
clima = aval[aval["Tipo_Avaliacao"] == "Clima_Organizacional"]
total = len(clima)
prom  = (clima["Nota_eNPS"] >= 9).sum()
det   = (clima["Nota_eNPS"] <= 6).sum()
enps  = round((prom - det) / total * 100, 2) if total > 0 else 0
print(f"eNPS: {enps}")

Absenteísmo
Benchmark: ≤ 3,5% · Fonte: AIHR
Fórmula: Absenteísmo (%) = (Dias ausentes / Dias úteis disponíveis) × 100
Dias úteis disponíveis = Dias úteis no período × Nº de colaboradores
Excel
=SOMASES(Faltas[Dias_Ausentes], Faltas[Departamento],A2, Faltas[Mes],B2)
/ (DIATRABALHOTOTAL(Data_Inicio, Data_Fim) * Headcount) * 100
DAX
Dias Ausentes =
CALCULATE(SUM(fato_Historico[Dias_Ausentes]), fato_Historico[Tipo_Evento] = "Ausência")

Absenteísmo % = DIVIDE([Dias Ausentes],
    NETWORKDAYS(MIN(dim_dCalendario[Date]),MAX(dim_dCalendario[Date])) * [HC Ativo],
    0) * 100
SQL
SELECT Departamento,
    ROUND(SUM(Dias_Ausentes)*100.0
          /((SELECT COUNT(*) FROM dim_Colaboradores WHERE Status='Ativo')*22),2) AS absenteismo_pct
FROM fato_Historico WHERE Tipo_Evento='Ausência' GROUP BY Departamento;
Python
ausencias     = hist[hist["Tipo_Evento"]=="Ausência"]
total_ausente = ausencias["Dias_Ausentes"].sum()
dias_uteis    = 22 * len(col[col["Status"]=="Ativo"])
absenteismo   = round(total_ausente / dias_uteis * 100, 2) if dias_uteis > 0 else 0
print(f"Absenteísmo: {absenteismo}%")

Time to Fill
Benchmark: 30–45 dias · Fonte: SHRM
Fórmula: TTF = Data da oferta aceita − Data de abertura da vaga (dias corridos)
Fonte: dim_Vagas filtrado por Motivo_Encerramento = 'Contratação'
Excel
=MÉDIASES(Vagas[Time_to_Fill], Vagas[Motivo_Encerramento],"Contratação", Vagas[Departamento],A2)
DAX
TTF Médio (dias) =
AVERAGEX(
    FILTER(dim_Vagas, dim_Vagas[Motivo_Encerramento] = "Contratação"),
    DATEDIFF(dim_Vagas[Data_Abertura], dim_Vagas[Data_Aceite], DAY)
)
SQL
SELECT Departamento,
    ROUND(AVG(JULIANDAY(Data_Aceite) - JULIANDAY(Data_Abertura)), 1) AS ttf_medio
FROM dim_Vagas WHERE Motivo_Encerramento = 'Contratação'
GROUP BY Departamento ORDER BY ttf_medio DESC;
Python
vagas  = pd.read_excel("workforce.xlsx", sheet_name="Vagas",
                       parse_dates=["Data_Abertura","Data_Aceite"])
filled = vagas[vagas["Motivo_Encerramento"]=="Contratação"].copy()
filled["ttf"] = (filled["Data_Aceite"] - filled["Data_Abertura"]).dt.days
print(f"TTF médio: {filled['ttf'].mean():.1f} dias")

Offer Acceptance Rate
Benchmark: > 85% · Fonte: Michael Page
Fórmula: OAR (%) = (Ofertas aceitas / Ofertas feitas) × 100
Proxy: se o ATS não rastreia ofertas, use Motivo_Encerramento = 'Contratação' como aceita e Oferta_Feita = 'S' como denominador.
Excel
=CONT.SES(Vagas[Oferta_Feita],"S",Vagas[Motivo_Encerramento],"Contratação")
/ CONT.SE(Vagas[Oferta_Feita],"S") * 100
DAX
Offer Acceptance Rate % =
DIVIDE(
    CALCULATE(COUNTROWS(dim_Vagas), dim_Vagas[Motivo_Encerramento]="Contratação", dim_Vagas[Oferta_Feita]="S"),
    CALCULATE(COUNTROWS(dim_Vagas), dim_Vagas[Oferta_Feita]="S"), 0
) * 100
SQL
SELECT ROUND(
    SUM(CASE WHEN Motivo_Encerramento='Contratação' THEN 1.0 ELSE 0 END)
    / NULLIF(SUM(CASE WHEN Oferta_Feita='S' THEN 1.0 ELSE 0 END),0) * 100, 2
) AS oar_pct FROM dim_Vagas;
Python
ofertas_feitas  = vagas[vagas["Oferta_Feita"]=="S"].shape[0]
ofertas_aceitas = vagas[(vagas["Oferta_Feita"]=="S") & (vagas["Motivo_Encerramento"]=="Contratação")].shape[0]
oar = round(ofertas_aceitas / ofertas_feitas * 100, 2) if ofertas_feitas > 0 else 0
print(f"Offer Acceptance Rate: {oar}%")

Cost per Hire
Benchmark: R$ 3.000–8.000 · Fonte: SHRM
Fórmula: CPH = (Custos internos + Custos externos) / Nº de contratações
Internos: horas do recrutador, tempo do gestor · Externos: job boards, agências, assessments
Excel
=SOMASES(Vagas[Custo_Total_Contratacao],Vagas[Motivo_Encerramento],"Contratação")
/CONT.SE(Vagas[Motivo_Encerramento],"Contratação")
DAX
Cost per Hire =
DIVIDE(
    CALCULATE(SUM(dim_Vagas[Custo_Total_Contratacao]), dim_Vagas[Motivo_Encerramento]="Contratação"),
    CALCULATE(COUNTROWS(dim_Vagas),                    dim_Vagas[Motivo_Encerramento]="Contratação"), 0
)
SQL
SELECT Departamento,
    ROUND(AVG(Custo_Total_Contratacao),2) AS cph_medio,
    COUNT(*)                              AS total_contratacoes
FROM dim_Vagas WHERE Motivo_Encerramento='Contratação'
GROUP BY Departamento ORDER BY cph_medio DESC;
Python
contratados = vagas[vagas["Motivo_Encerramento"]=="Contratação"]
cph = round(contratados["Custo_Total_Contratacao"].mean(), 2)
print(f"Cost per Hire médio: R$ {cph:,.2f}")

Quality of Hire
Benchmark: > 70/100 · Fonte: AIHR
Fórmula: Quality of Hire = (Performance Score + Retention Score) / 2 · Ambos em escala 0–100.
Media_Gestor (0–5) → multiplique × 20 para converter para 0–100. Retention Score: 100 se ativo após 12 meses; 0 caso contrário.
DAX
Performance Score (0-100) =
AVERAGEX(FILTER(fato_Avaliacoes, fato_Avaliacoes[Tipo_Avaliacao]="Avaliacao_180"),
         fato_Avaliacoes[Media_Gestor] * 20)

Retention Score (0-100) =
DIVIDE(
    CALCULATE(COUNTROWS(dim_Colaboradores), dim_Colaboradores[Status]="Ativo",
              DATEDIFF(dim_Colaboradores[Data_Admissao], TODAY(), MONTH) >= 12),
    CALCULATE(COUNTROWS(dim_Colaboradores), DATEDIFF(dim_Colaboradores[Data_Admissao], TODAY(), MONTH) >= 12),
    0) * 100

Quality of Hire = DIVIDE([Performance Score (0-100)] + [Retention Score (0-100)], 2)
SQL
WITH perf AS (
    SELECT ID_Colaborador, AVG(Media_Gestor * 20) AS perf_score
    FROM fato_Avaliacoes WHERE Tipo_Avaliacao = 'Avaliacao_180' GROUP BY 1
),
ret AS (
    SELECT ID_Colaborador,
           CASE WHEN Status='Ativo' AND JULIANDAY('now')-JULIANDAY(Data_Admissao)>=365
                THEN 100 ELSE 0 END AS ret_score
    FROM dim_Colaboradores WHERE JULIANDAY('now')-JULIANDAY(Data_Admissao)>=365
)
SELECT ROUND(AVG((p.perf_score + r.ret_score)/2.0),2) AS quality_of_hire
FROM ret r JOIN perf p ON r.ID_Colaborador = p.ID_Colaborador;
Python
hoje = pd.Timestamp(date.today())
elegivel = col[((hoje-col["Data_Admissao"])/pd.Timedelta(days=365))>=1].copy()
elegivel["ret_score"] = (elegivel["Status"]=="Ativo").astype(int)*100
aval_180 = aval[aval["Tipo_Avaliacao"]=="Avaliacao_180"].groupby("ID_Colaborador")["Media_Gestor"].mean()*20
elegivel = elegivel.merge(aval_180.rename("perf_score"), on="ID_Colaborador", how="left")
elegivel["qoh"] = (elegivel["perf_score"].fillna(0) + elegivel["ret_score"]) / 2
print(f"Quality of Hire médio: {elegivel['qoh'].mean():.1f}")

)
SQL
SELECT ROUND(AVG(JULIANDAY(Data_Aceite)-JULIANDAY(Data_Candidatura_Contratado)),1) AS tth_medio
FROM dim_Vagas WHERE Motivo_Encerramento='Contratação' AND Data_Candidatura_Contratado IS NOT NULL;
Python
filled["tth"] = (filled["Data_Aceite"] - filled["Data_Candidatura_Contratado"]).dt.days

Por que nosso turnover está acima da média do setor? +

Diagnóstico em 3 camadas:

  1. Camada demográfica — segmente o turnover por departamento, faixa etária, tempo de casa (0–6 m, 7–12 m, 1–3 a, 3+ a) e gestor. Um turnover de 25% pode esconder um departamento em 60% e outro em 5%.
  2. Camada temporal — plote saídas mês a mês. Picos sazonais (jan-fev pós-bônus, jul-ago) indicam retenção por incentivo financeiro, não por engajamento real.
  3. Camada de motivo — cruze fato_Historico[Motivo] com data de saída. Se "proposta externa" domina, o problema é competitividade salarial; se "clima/gestão" domina, o problema é liderança.
DAX — Turnover por Gestor (Top 5 piores)
Turnover por Gestor =
VAR saidas =
    CALCULATE(
        COUNTROWS(fato_Historico),
        fato_Historico[Tipo_Evento] = "Desligamento"
    )
VAR hc_medio = [HC Médio]
RETURN DIVIDE(saidas, hc_medio, 0) * 100

Crie um visual de tabela com dim_Colaboradores[Gestor_Direto] no eixo e esta medida como valor. Ordene decrescente e aplique formatação condicional: ≥20% vermelho, 10–20% amarelo.

Ação recomendada: Apresente ao CHRO o "mapa de calor de turnover" antes de propor qualquer intervenção. Evite soluções universais (ex.: aumento geral de salário) quando o problema é localizado.

Quais colaboradores têm maior risco de saída nos próximos 90 dias? +

Modelo de Flight Risk com 5 sinais:

SinalComo medirPeso sugerido
eNPS individual ≤ 6fato_Avaliacoes[Nota_eNPS]30%
Avaliação de desempenho ≤ 2/5 ou ≥ 4.5/5fato_Avaliacoes[Media_Gestor]20%
Tempo no cargo atual > 18 meses sem promoçãodim_Colaboradores[Data_Ultima_Promocao]20%
Gestor com turnover histórico > 20%Medida derivada15%
Salário abaixo da mediana da faixadim_Colaboradores[Salario] vs banda15%
Python — Score de Flight Risk
import pandas as pd
from datetime import date

df = pd.read_excel("workforce.xlsx", parse_dates=["Data_Admissao","Data_Ultima_Promocao"])
hoje = pd.Timestamp(date.today())

df["meses_sem_promocao"] = ((hoje - df["Data_Ultima_Promocao"]) / pd.Timedelta(days=30)).round()
df["enps_baixo"]         = (df["Nota_eNPS"] <= 6).astype(int)
df["sem_promocao_flag"]  = (df["meses_sem_promocao"] >= 18).astype(int)
df["perf_extrema"]       = ((df["Media_Gestor"] <= 2) | (df["Media_Gestor"] >= 4.5)).astype(int)

df["flight_risk_score"] = (
    df["enps_baixo"]        * 30 +
    df["perf_extrema"]      * 20 +
    df["sem_promocao_flag"] * 20
) / 70  # normaliza 0-100

alto_risco = df[df["flight_risk_score"] >= 60].sort_values("flight_risk_score", ascending=False)
print(alto_risco[["Nome","Departamento","Gestor","flight_risk_score"]].head(20))

Atenção LGPD: Este modelo processa dados sensíveis. Garanta que o propósito esteja documentado na ROPA, que o acesso seja restrito ao HRBP e ao gestor imediato, e que o colaborador não seja comunicado sem uma ação concreta de retenção vinculada.

Nosso processo seletivo está demorando demais — onde está o gargalo? +

Decomponha o Time to Fill em etapas:

SQL — Tempo médio por etapa do funil
-- Requer colunas de data por etapa em dim_Vagas
SELECT
    Departamento,
    AVG(JULIANDAY(Data_Triagem)      - JULIANDAY(Data_Abertura))    AS dias_abertura_triagem,
    AVG(JULIANDAY(Data_Entrevista)   - JULIANDAY(Data_Triagem))     AS dias_triagem_entrevista,
    AVG(JULIANDAY(Data_Oferta)       - JULIANDAY(Data_Entrevista))  AS dias_entrevista_oferta,
    AVG(JULIANDAY(Data_Aceite)       - JULIANDAY(Data_Oferta))      AS dias_oferta_aceite,
    AVG(JULIANDAY(Data_Aceite)       - JULIANDAY(Data_Abertura))    AS time_to_fill_total
FROM dim_Vagas
WHERE Motivo_Encerramento = 'Contratação'
GROUP BY Departamento
ORDER BY time_to_fill_total DESC;

Interpretação rápida:

  • Gargalo em triagem→entrevista > 7 dias: Recrutador sobrecarregado ou critérios de triagem ambíguos → revisar scorecard de triagem.
  • Gargalo em entrevista→oferta > 5 dias: Aprovação interna lenta (comitê, VP) → criar alçada de aprovação simplificada para vagas operacionais.
  • Gargalo em oferta→aceite > 3 dias: Proposta abaixo da expectativa ou concorrência ativa → benchmark salarial antes da oferta.

Meta de benchmark: Time to Fill total ≤ 30 dias (individual contributors), ≤ 45 dias (gestores/especialistas sênior). Fonte: SHRM Talent Acquisition Benchmarking Report.

Como demonstrar o ROI de uma iniciativa de treinamento? +

Modelo de ROI de L&D (Kirkpatrick + Phillips):

  1. Nível 1 — Reação: Pesquisa pós-treinamento (NPS do programa). Meta: ≥ 70.
  2. Nível 2 — Aprendizado: Pré-teste vs. pós-teste. Meta: ganho ≥ 20 pontos percentuais.
  3. Nível 3 — Comportamento: Avaliação 90 dias depois pelo gestor. Use fato_Avaliacoes com Tipo_Avaliacao = 'Pos_Treinamento'.
  4. Nível 4 — Resultado: Reduções em turnover, aumento em produtividade, queda em erros operacionais — mensuráveis em KPIs já rastreados.
Fórmula — ROI de Treinamento
ROI (%) = ((Benefício Líquido - Custo Total do Programa) / Custo Total do Programa) × 100

Exemplo:
  Custo do programa de liderança: R$ 80.000
  Redução de turnover de gestores (3 saídas evitadas × R$ 40.000 custo médio de substituição):
      Benefício = 3 × R$ 40.000 = R$ 120.000
  ROI = ((120.000 - 80.000) / 80.000) × 100 = 50%

Dica: O custo de substituição de um colaborador varia de 50% (cargos operacionais) a 200% (cargos técnicos/gestão) do salário anual. Use o valor conservador (50%) para cálculos de ROI — o argumento fica mais robusto se a conta "pessimista" ainda justifica o investimento.

Como calcular o custo real de uma contratação mal feita? +

Componentes do custo de uma contratação mal feita:

ComponenteEstimativa típica
Custo de Recrutamento (CPH)R$ 3.000–15.000
Onboarding e treinamento inicial1–3 meses de salário
Produtividade perdida (ramp-up)3–6 meses de salário
Impacto na equipe (moral, retrabalho)20–30% da produtividade do time por 1–3 meses
Nova contratação (se demitido)50–200% do salário anual
Total conservador1,5× a 3× o salário anual do cargo
Estimativa rápida
Custo de Contratação Mal Feita =
    CPH
    + (Salário_Mensal × Meses_Treinamento)
    + (Salário_Mensal × Meses_Produtividade_Perdida × 0.5)
    + CPH  -- novo processo seletivo

Exemplo (Analista Sênior, salário R$ 8.000/mês):
    CPH = R$ 6.000
    Treinamento = 8.000 × 2 = R$ 16.000
    Produtividade = 8.000 × 4 × 0.5 = R$ 16.000
    Nova busca = R$ 6.000
    TOTAL = R$ 44.000 (≈ 5,5 salários mensais)

Use este número para justificar investimento em Quality of Hire: um processo seletivo mais robusto que custa R$ 10 K a mais mas evita uma contratação mal feita de R$ 44 K tem ROI de 340%.

Nosso eNPS caiu 15 pontos — como investigar e agir? +

Protocolo de investigação de queda de eNPS:

  1. Segmentar antes de agir: A queda é generalizada ou concentrada em área/nível/gestor específico? Uma queda de 15 pontos na empresa pode ser 40 pontos em uma área e 0 nas demais.
  2. Cruzar com dados quantitativos: Houve aumento de turnover, aumento de absenteísmo ou queda de performance no mesmo período? Correlações confirmam ou descartam hipóteses.
  3. Focus groups com detratores (anonimizados): eNPS mede intensidade, não motivo. Use entrevistas qualitativas com uma amostra representativa de detratores (6–8 pessoas) para entender o "porquê".
  4. Plano de ação em 30 dias: Comunique à empresa que os dados foram analisados e que há um plano. O silêncio após uma pesquisa é um dos maiores destruidores de confiança em RH.
DAX — eNPS por Departamento (variação vs. período anterior)
eNPS Atual =
VAR base  = FILTER(fato_Avaliacoes, fato_Avaliacoes[Tipo_Avaliacao] = "Clima_Organizacional")
VAR total = COUNTROWS(base)
VAR prom  = COUNTROWS(FILTER(base, fato_Avaliacoes[Nota_eNPS] >= 9))
VAR det   = COUNTROWS(FILTER(base, fato_Avaliacoes[Nota_eNPS] <= 6))
RETURN DIVIDE(prom - det, total, 0) * 100

eNPS Período Anterior =
CALCULATE([eNPS Atual], DATEADD(dim_dCalendario[Date], -6, MONTH))

Variação eNPS = [eNPS Atual] - [eNPS Período Anterior]

Benchmark de ação: eNPS ≥ 30 = saudável; 0–29 = zona de melhoria; abaixo de 0 = crítico — ação imediata necessária.

dim_Colaboradores[ID_Colaborador] → fato_Historico[ID_Colaborador] (1:N)
dim_Colaboradores[ID_Colaborador] → fato_Avaliacoes[ID_Colaborador] (1:N)
DAX — Tabela de Calendário (cole no Power Query avançado)
dim_dCalendario =
VAR DataInicio = DATE(2020, 1, 1)
VAR DataFim    = TODAY()
RETURN
ADDCOLUMNS(
    CALENDAR(DataInicio, DataFim),
    "Ano",         YEAR([Date]),
    "Mês",         MONTH([Date]),
    "NomeMês",     FORMAT([Date], "MMMM"),
    "Trimestre",   "T" & QUARTER([Date]),
    "AnoMês",      FORMAT([Date], "YYYY-MM"),
    "DiaSemana",   WEEKDAY([Date], 2)  -- 1=Seg, 7=Dom
)
Boas práticas Power BI para People Analytics:
  • Use RLS (Row-Level Security) para que gestores vejam apenas os dados da sua equipe
  • Defina uma tabela de medidas separada (_Medidas) para organizar os KPIs
  • Prefira DIVIDE(numerador, denominador, 0) a divisão direta — evita erros de divisão por zero
  • Documente cada medida com Description no campo de propriedades

Python — Pipeline de ETL para People Analytics

people_analytics_pipeline.py — Esqueleto completo
import pandas as pd
from datetime import date

# ── 1. EXTRAÇÃO ──────────────────────────────────────────────
def extrair_dados(caminho_excel: str) -> dict[str, pd.DataFrame]:
    """Lê todas as abas do arquivo de RH."""
    return pd.read_excel(
        caminho_excel,
        sheet_name=["Colaboradores", "Historico", "Avaliacoes", "Vagas"],
        parse_dates=True
    )

# ── 2. TRANSFORMAÇÃO ─────────────────────────────────────────
def calcular_kpis(dados: dict) -> pd.DataFrame:
    hoje = pd.Timestamp(date.today())
    col  = dados["Colaboradores"]
    hist = dados["Historico"]
    aval = dados["Avaliacoes"]

    # Headcount atual
    hc_atual = col[col["Status"] == "Ativo"].shape[0]

    # Turnover Rate (período: ano corrente)
    ano_atual = hoje.year
    saidas = hist[
        (hist["Tipo_Evento"] == "Desligamento") &
        (hist["Data_Evento"].dt.year == ano_atual)
    ].shape[0]

    hc_inicio = col[col["Data_Admissao"].dt.year < ano_atual].shape[0]
    hc_medio  = (hc_inicio + hc_atual) / 2
    turnover  = round((saidas / hc_medio * 100), 2) if hc_medio > 0 else 0

    # eNPS
    clima = aval[aval["Tipo_Avaliacao"] == "Clima_Organizacional"]
    total = len(clima)
    prom  = (clima["Nota_eNPS"] >= 9).sum()
    det   = (clima["Nota_eNPS"] <= 6).sum()
    enps  = round(((prom - det) / total * 100), 2) if total > 0 else 0

    return pd.DataFrame([{
        "Data":          hoje.date(),
        "HC_Atual":      hc_atual,
        "Turnover_Pct":  turnover,
        "eNPS":          enps
    }])

# ── 3. CARGA ─────────────────────────────────────────────────
def salvar_resultado(df: pd.DataFrame, destino: str) -> None:
    df.to_excel(destino, index=False)
    print(f"KPIs salvos em: {destino}")

# ── EXECUÇÃO ─────────────────────────────────────────────────
if __name__ == "__main__":
    dados   = extrair_dados("workforce.xlsx")
    kpis    = calcular_kpis(dados)
    salvar_resultado(kpis, "kpis_people_analytics.xlsx")

SQL — Queries de referência para People Analytics

View consolidada de KPIs mensais
CREATE VIEW vw_kpis_people_analytics AS
WITH
  -- Headcount médio por mês
  hc_mensal AS (
    SELECT
      strftime('%Y-%m', Data_Evento) AS ano_mes,
      COUNT(DISTINCT ID_Colaborador)  AS hc
    FROM fato_Historico
    WHERE Tipo_Evento IN ('Admissao','Ativo')
    GROUP BY 1
  ),
  -- Saídas por mês
  saidas_mensais AS (
    SELECT
      strftime('%Y-%m', Data_Evento) AS ano_mes,
      COUNT(*)                        AS saidas,
      SUM(CASE WHEN Motivo = 'Voluntário' THEN 1 ELSE 0 END) AS saidas_voluntarias
    FROM fato_Historico
    WHERE Tipo_Evento = 'Desligamento'
    GROUP BY 1
  ),
  -- eNPS por mês
  enps_mensal AS (
    SELECT
      strftime('%Y-%m', Data_Avaliacao) AS ano_mes,
      ROUND(
        (SUM(CASE WHEN Nota_eNPS >= 9 THEN 1.0 ELSE 0 END) -
         SUM(CASE WHEN Nota_eNPS <= 6 THEN 1.0 ELSE 0 END))
        / COUNT(*) * 100, 2
      ) AS enps
    FROM fato_Avaliacoes
    WHERE Tipo_Avaliacao = 'Clima_Organizacional'
    GROUP BY 1
  )
SELECT
  h.ano_mes,
  h.hc                                              AS headcount,
  s.saidas,
  ROUND(s.saidas * 100.0 / h.hc, 2)                AS turnover_pct,
  ROUND(s.saidas_voluntarias * 100.0 / h.hc, 2)    AS vol_turnover_pct,
  e.enps
FROM hc_mensal h
LEFT JOIN saidas_mensais s ON h.ano_mes = s.ano_mes
LEFT JOIN enps_mensal    e ON h.ano_mes = e.ano_mes
ORDER BY h.ano_mes;

Fórmulas-chave na aba KPIs_Calculados
-- Headcount atual (coluna B = Status)
=CONT.SE(Dados_Colaboradores[Status],"Ativo")

-- Turnover Rate (coluna C = Tipo_Evento, coluna E = Data_Evento)
=CONT.SES(
    Dados_Historico[Tipo_Evento], "Desligamento",
    Dados_Historico[Data_Evento], ">="&DATA(ANO(HOJE()),1,1),
    Dados_Historico[Data_Evento], "<="&HOJE()
) / ((B2 + B1) / 2) * 100
-- B2 = HC atual; B1 = HC início do ano (calculado separado)

-- eNPS (coluna F = Nota_eNPS, coluna G = Tipo_Avaliacao)
=(CONT.SES(Dados_Avaliacoes[Tipo_Avaliacao],"Clima_Organizacional",
           Dados_Avaliacoes[Nota_eNPS],">="&9)
 -CONT.SES(Dados_Avaliacoes[Tipo_Avaliacao],"Clima_Organizacional",
           Dados_Avaliacoes[Nota_eNPS],"<="&6))
/ CONT.SE(Dados_Avaliacoes[Tipo_Avaliacao],"Clima_Organizacional")
* 100

-- Time to Fill médio (dias)
=MÉDIASES(
    Dados_Vagas[Time_to_Fill],
    Dados_Vagas[Motivo_Encerramento], "Contratação"
)
